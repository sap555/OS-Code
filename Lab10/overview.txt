b) struct HDD {
    int cylinders;
    int tracks_per_cylinder;
    int sectors_per_track;
    int bytes_per_sector;
    int rpm;

    long long totalCapacity() {
        return (long long)cylinders *
               tracks_per_cylinder *
               sectors_per_track *
               bytes_per_sector;
    }
};
